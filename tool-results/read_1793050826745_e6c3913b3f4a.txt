    53→function EvaluationGraph({ points, currentMove, onSelectMove }) {
    54→  if (!points?.length) {
    55→    return <div className="graph-empty">Run a full review to see the game’s evaluation trend.</div>;
    56→  }
    57→  const width = 600;
    58→  const height = 220;
    59→  const pad = 28;
    60→  const rawValues = points.map((point) => Number.isFinite(point) ? point / 100 : 0);
    61→  const maxAbsValue = Math.max(0, ...rawValues.map((value) => Math.abs(value)));
    62→  const range = Math.min(12, Math.max(1, Math.ceil(maxAbsValue * 1.12)));
    63→  const values = rawValues.map((value) => Math.max(-range, Math.min(range, value)));
    64→  const x = (index) => pad + (points.length < 2 ? 0 : (index / (points.length - 1)) * (width - pad * 2));
    65→  const y = (value) => pad + ((range - value) / (range * 2)) * (height - pad * 2);
    66→  const curvePoints = values.map((value, index) => ({ x: x(index), y: y(value) }));
    67→  const curveSegments = curvePoints.slice(0, -1).map((point, index) => {
    68→    const previous = curvePoints[Math.max(0, index - 1)];
    69→    const next = curvePoints[index + 1];
    70→    const afterNext = curvePoints[Math.min(curvePoints.length - 1, index + 2)];
    71→    const control1 = { x: point.x + (next.x - previous.x) / 6, y: point.y + (next.y - previous.y) / 6 };
    72→    const control2 = { x: next.x - (afterNext.x - point.x) / 6, y: next.y - (afterNext.y - point.y) / 6 };
    73→    return `C ${control1.x} ${control1.y}, ${control2.x} ${control2.y}, ${next.x} ${next.y}`;
    74→  }).join(" ");
    75→  const line = `M ${curvePoints[0].x} ${curvePoints[0].y} ${curveSegments}`;
    76→  const areaAbove = `M ${curvePoints[0].x} ${pad} L ${curvePoints[0].x} ${curvePoints[0].y} ${curveSegments} L ${curvePoints[curvePoints.length - 1].x} ${pad} Z`;
    77→  const areaBelow = `M ${curvePoints[0].x} ${height - pad} L ${curvePoints[0].x} ${curvePoints[0].y} ${curveSegments} L ${curvePoints[curvePoints.length - 1].x} ${height - pad} Z`;
    78→  const zeroY = y(0);
    79→  return (
    80→    <div className="eval-graph-wrap">
    81→      <div className="graph-legend"><span><i className="white-dot" /> White advantage</span><span><i className="black-dot" /> Black advantage</span></div>
    82→      <svg className="eval-graph" viewBox={`0 0 ${width} ${height}`} role="img" aria-label="Evaluation through the game">
    83→        <rect className="graph-plot-surface" x={pad} y={pad} width={width - pad * 2} height={height - pad * 2} rx="8" />
    84→        <defs><clipPath id="evaluation-plot-clip"><rect x={pad} y={pad} width={width - pad * 2} height={height - pad * 2} rx="8" /></clipPath></defs>
    85→        <g clipPath="url(#evaluation-plot-clip)">
    86→          <path d={areaAbove} className="graph-white-area" />
    87→          <path d={areaBelow} className="graph-black-area" />
    88→          <line x1={pad} y1={zeroY} x2={width - pad} y2={zeroY} className="graph-zero" />
    89→          <path d={line} className="graph-line" />
    90→          {values.map((value, index) => (
    91→            <circle key={index} cx={x(index)} cy={y(value)} r={index === currentMove ? 5 : 2.4}
    92→              className={index === currentMove ? "graph-point current" : "graph-point"}
    93→              role="button" tabIndex={0} aria-label={index === 0 ? "Start position" : `Move ${index}`}
    94→              onClick={() => onSelectMove(index)} onKeyDown={(event) => event.key === "Enter" && onSelectMove(index)}>
    95→              <title>{index === 0 ? "Start position" : `After move ${index}`} · {value >= 0 ? `White +${value.toFixed(2)}` : `Black +${Math.abs(value).toFixed(2)}`} pawns</title>
    96→            </circle>
    97→          ))}
    98→        </g>
    99→        <text x={pad - 5} y={pad - 9} textAnchor="start" className="graph-y-label">+{range.toFixed(1)}</text>
   100→        <text x={pad - 5} y={height - pad + 15} textAnchor="start" className="graph-y-label">−{range.toFixed(1)}</text>
   101→      </svg>
   102→      <div className="graph-axis"><span>Start</span><span>Move {points.length - 1}</span></div>
   103→    </div>
   104→  );
   105→}

   701→  const materialBalanceFor = (chess, color) => chess.board().flat().reduce((balance, piece) => {
   702→    if (!piece || piece.type === "k") return balance;

   707→  const findMaterialSacrifice = (previousFen, playedMove) => {
   708→    try {
   709→      const chess = new Chess(previousFen);
   710→      const mover = chess.turn();
   711→      const materialBefore = materialBalanceFor(chess, mover);
   712→      const move = chess.move({
   713→        from: playedMove.slice(0, 2),
   714→        to: playedMove.slice(2, 4),
   715→        promotion: playedMove[4] || "q",
   716→      });
   717→
   718→      if (!move) return null;
   719→      const movedPiece = chess.get(move.to);
   720→      const movedPieceValue = movedPiece ? PIECE_VALUES[movedPiece.type] || 0 : 0;
   721→      if (movedPieceValue < 300) return null;
   722→
   723→      // Follow the complete legal exchange on the destination square. Looking
   724→      // only one recapture deep can miss a real sacrifice when the exchange
   725→      // continues and the tactical compensation appears later in the line.
   726→      const exchangeMemo = new Map();
   727→      const bestExchangeBalance = (position, targetSquare) => {
   728→        const key = `${position.fen()}|${targetSquare}|${mover}`;
   729→        if (exchangeMemo.has(key)) return exchangeMemo.get(key);
   730→        const standPat = materialBalanceFor(position, mover);
   731→        const captures = position.moves({ verbose: true }).filter((reply) => reply.to === targetSquare && reply.captured);
   732→        if (!captures.length) {
   733→          exchangeMemo.set(key, standPat);
   734→          return standPat;
   735→        }
   736→
   737→        const balances = captures.map((capture) => {
   738→          const next = new Chess(position.fen());
   739→          next.move({ from: capture.from, to: capture.to, promotion: capture.promotion });
   740→          return bestExchangeBalance(next, targetSquare);
   741→        });
   742→        const moverChooses = position.turn() === mover;
   743→        const continuation = moverChooses ? Math.max(...balances) : Math.min(...balances);
   744→        const bestBalance = moverChooses ? Math.max(standPat, continuation) : Math.min(standPat, continuation);
   745→        exchangeMemo.set(key, bestBalance);
   746→        return bestBalance;
   747→      };
   748→
   749→      // A piece being attacked does not by itself make a sacrifice. The
   750→      // opponent must have a legal capture that leaves a meaningful net loss
   751→      // after both sides choose their best continuation in that exchange.
   752→      const captures = chess.moves({ verbose: true }).filter((reply) => reply.to === move.to && reply.captured);
   753→      for (const capture of captures) {
   754→        const afterCapture = new Chess(chess.fen());
   755→        afterCapture.move({ from: capture.from, to: capture.to, promotion: capture.promotion });
   756→        const bestMaterialAfterReply = bestExchangeBalance(afterCapture, move.to);
   757→        const netMaterialOffered = materialBefore - bestMaterialAfterReply;
   758→        if (netMaterialOffered >= 200) return { netMaterialOffered, offeredPieceValue: movedPieceValue };
   759→      }
   760→      return null;
   761→    } catch {
   762→      return null;
   763→    }

   789→  const classifyMoveReview = async ({

   876→  const analyzeFenOnce = (fenToAnalyze, depth, forcedMove = null, multiPv = 1) => {
   877→    return new Promise((resolve, reject) => {
   878→      const worker = new Worker(STOCKFISH_PATH);
   879→
   880→      const scoresByLine = {};
   881→      const movesByDepth = {};
   882→      const requestedTopMoveCount = Math.max(1, Math.min(multiPv, new Chess(fenToAnalyze).moves().length));
   883→      let latestBestMove = null;
   884→      let finished = false;
   885→
   886→      const finish = (result) => {
   887→        if (finished) return;
   888→        finished = true;
   889→        worker.terminate();
   890→        resolve(result);
   891→      };
   892→
   893→      const timeout = setTimeout(() => {
   894→        if (!finished) {
   895→          worker.terminate();
   896→          reject(new Error("Stockfish analysis timed out"));
   897→        }
   898→      }, depth > 16 ? 60000 : 30000);
   899→
   900→      worker.onmessage = (event) => {
   901→        const line = String(event.data);
   902→
   903→        if (line === "uciok") {
   904→          worker.postMessage("setoption name UCI_ShowWDL value true");
   905→          worker.postMessage(`setoption name MultiPV value ${multiPv}`);
   906→          worker.postMessage("isready");
   907→          return;
   908→        }
   909→
   910→        if (line === "readyok") {
   911→          worker.postMessage(`position fen ${fenToAnalyze}`);
   912→          worker.postMessage(
   913→            forcedMove
   914→              ? `go depth ${depth} searchmoves ${forcedMove}`
   915→              : `go depth ${depth}`
   916→          );
   917→          return;
   918→        }
   919→
   920→        if (line.startsWith("info depth")) {
   921→          const depthMatch = line.match(/\bdepth (\d+)/);
   922→          const multipvMatch = line.match(/\bmultipv (\d+)/);
   923→          const lineDepth = depthMatch ? Number(depthMatch[1]) : 0;
   924→          const lineNumber = multipvMatch ? Number(multipvMatch[1]) : 1;
   925→          const cpMatch = line.match(/score cp (-?\d+)/);
   926→          const mateMatch = line.match(/score mate (-?\d+)/);
   927→          const wdlMatch = line.match(/\bwdl\s+(\d+)\s+(\d+)\s+(\d+)/i);
   928→          const pvMatch = line.match(/ pv ([a-h][1-8][a-h][1-8][qrbn]?)/);
   929→          const wdl = wdlMatch ? wdlMatch.slice(1).map(Number) : null;
   930→          const isBound = /\b(?:lowerbound|upperbound)\b/.test(line);
   931→
   932→          let score = null;
   933→          if (mateMatch) {
   934→            score = {
   935→              type: "mate",
   936→              value: Number(mateMatch[1]),
   937→              wdl,
   938→            };
   939→          } else if (cpMatch) {
   940→            score = {
   941→              type: "cp",
   942→              value: Number(cpMatch[1]),
   943→              wdl,
   944→            };
   945→          }
   946→
   947→          if (score && !isBound && lineDepth >= (scoresByLine[lineNumber]?.depth || 0)) {
   948→            scoresByLine[lineNumber] = { depth: lineDepth, score };
   949→            if (pvMatch) {
   950→              movesByDepth[lineDepth] ||= {};
   951→              movesByDepth[lineDepth][lineNumber] = pvMatch[1];
   952→            }
   953→            if (lineNumber === 1 && pvMatch) latestBestMove = pvMatch[1];
   954→          }
   955→          return;
   956→        }
   957→
   958→        if (line.startsWith("bestmove")) {
   959→          clearTimeout(timeout);
   960→          const requiredLineNumbers = Array.from({ length: requestedTopMoveCount }, (_, index) => index + 1);
   961→          const deepestCompleteLines = Object.keys(movesByDepth)
   962→            .map(Number)
   963→            .sort((a, b) => b - a)
   964→            .map((lineDepth) => movesByDepth[lineDepth])
   965→            .find((lines) => requiredLineNumbers.every((lineNumber) => lines[lineNumber]));
   966→          const topMoves = deepestCompleteLines
   967→            ? requiredLineNumbers.map((lineNumber) => deepestCompleteLines[lineNumber])
   968→            : [];
   969→
   970→          const move = line.split(" ")[1];
   971→
   972→          finish({
   973→            score: scoresByLine[1]?.score || null,
   974→            topMoves,
   975→            alternatives: Object.fromEntries(
   976→              Object.entries(scoresByLine).map(([lineNumber, lineInfo]) => [
   977→                lineNumber,
   978→                lineInfo.score,
   979→              ])
   980→            ),
   981→            depths: Object.fromEntries(
   982→              Object.entries(scoresByLine).map(([lineNumber, lineInfo]) => [
   983→                lineNumber,
   984→                lineInfo.depth,
   985→              ])
   986→            ),
   987→            bestMove: move || latestBestMove,
   988→          });
   989→        }
   990→      };
   991→
   992→      worker.onerror = (error) => {
   993→        clearTimeout(timeout);
   994→        worker.terminate();
   995→        reject(error);
   996→      };
   997→
   998→      worker.postMessage("uci");
   999→    });
  1000→  };

  1002→  const runFullReview = async () => {
  1003→    if (!uciMoveHistory.length || reviewState.status === "running") return;
  1004→    const runId = reviewRunRef.current + 1;
  1005→    reviewRunRef.current = runId;
  1006→    setReviewState({ status: "running", completed: 0, total: uciMoveHistory.length });
